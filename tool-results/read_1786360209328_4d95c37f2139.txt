    79→// ─── Helper: parse kopLines from API response ────────────────────────────────
    80→
    81→function parseKopLines(raw: unknown): KopLine[] {
    82→  if (!raw) return []
    83→  try {
    84→    const parsed = typeof raw === 'string' ? JSON.parse(raw) : raw
    85→    if (!Array.isArray(parsed)) return []
    86→    return parsed.map((item: unknown) => {
    87→      // Backward compatibility: if item is a string, treat as detail line
    88→      if (typeof item === 'string') {
    89→        return { text: item, style: 'detail' as const, bold: false }
    90→      }
    91→      if (typeof item === 'object' && item !== null) {
    92→        const obj = item as Record<string, unknown>
    93→        return {
    94→          text: String(obj.text ?? ''),
    95→          style: (obj.style === 'header' ? 'header' : 'detail') as 'header' | 'detail',
    96→          bold: Boolean(obj.bold ?? false),
    97→        }
    98→      }
    99→      return { text: '', style: 'detail' as const, bold: false }
   100→    }).filter((l: KopLine) => l.text.trim() !== '' || true) // keep empty lines for editing
   101→  } catch {
   102→    return []
   103→  }
   104→}
